async function loadNews(query) { const newsDiv = document.getElementById("news"); const status = document.getElementById("status"); newsDiv.innerHTML = ""; status.innerText = "جاري تحميل الأخبار..."; try { const res = await fetch( `https://api.gdeltproject.org/api/v2/doc/doc?query=${query}&mode=ArtList&format=json` ); const data = await res.json(); if (!data.articles || data.articles.length === 0) { status.innerText = "لا توجد أخبار"; return; } status.innerText = ""; data.articles.slice(0, 12).forEach(article => { const img = article.imageurl ? article.imageurl : "https://via.placeholder.com/400x200?text=No+Image"; const card = document.createElement("div"); card.className = "card"; card.innerHTML = `

${article.title}

${article.sourceCountry || "خبر عالمي"}

`; newsDiv.appendChild(card); }); } catch (err) { console.error(err); status.innerText = "❌ فشل تحميل الأخبار"; } }
